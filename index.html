O desafio do Luigi 🍕

Luigi tem uma pizzaria e precisa entregar uma pizza para cada um dos seus 5 amigos: sair da pizzaria, visitar a casa de cada amigo uma única vez, e voltar — sem deixar as pizzas esfriarem.

IloEnv env;
IloModel model(env);
IloArray<IloBoolVarArray> x(env, n);
for (IloInt i = 0; i < n; i++)
    x[i] = IloBoolVarArray(env, n);

IloExpr obj(env);
for (IloInt i = 0; i < n; i++)
  for (IloInt j = 0; j < n; j++)
    if (i != j) obj += c[i][j] * x[i][j];
model.add(IloMinimize(env, obj));

for (IloInt i = 0; i < n; i++) {
  IloExpr s1(env), s2(env);
  for (IloInt j = 0; j < n; j++) if (i != j) {
    s1 += x[i][j]; s2 += x[j][i];
  }
  model.add(s1 == 1);
  model.add(s2 == 1);
}
// (restricoes u_i - u_j + n*x_ij <= n-1 aqui...)

IloCplex cplex(model);
cplex.solve();
